[bug · medium]
`provider_models()` uses a single global cache (`_MODELS_CACHE`) without keying by provider identity (`base_url` and auth key). After runtime config/model changes, the function can return stale model IDs from a previous provider for up to 10 minutes, which affects lane verification and `/api/models` output.

**Suggestion:**
```suggestion
_MODELS_CACHE = {"t": 0.0, "models": None, "provider": None}

def provider_models():
    base = (CONFIG.get("base_url") or "").strip().rstrip("/")
    key = CONFIG.get("anthropic_key") or ""
    if not base or not key:
        return None
    provider = (base, hashlib.sha256(key.encode()).hexdigest()[:16])
    if _MODELS_CACHE.get("provider") == provider and time.time() - _MODELS_CACHE["t"] < 600:
        return _MODELS_CACHE["models"]
    try:
        req = urllib.request.Request(
            base + "/models",
            headers={"Authorization": f"Bearer {key}"})
        with urllib.request.urlopen(req, timeout=5) as r:
            data = json.load(r)
        ids = sorted(str(m.get("id")) for m in data.get("data", []) if m.get("id"))
        _MODELS_CACHE.update(t=time.time(), models=ids, provider=provider)
    except Exception:
        _MODELS_CACHE.update(t=time.time(), provider=provider)
    return _MODELS_CACHE["models"]
```
